#Run
gpg --homedir ~/labops/gnupg-release --fingerprint labops-release@labops.example


#Now run
gpg --homedir ~/labops/gnupg-attacker --fingerprint labops-release@labops.example


#You will get two different fingerprints.

#Save both:
{
echo "=== release key ==="
gpg --homedir ~/labops/gnupg-release --fingerprint labops-release@labops.example
echo "=== attacker key ==="
gpg --homedir ~/labops/gnupg-attacker --fingerprint labops-release@labops.example
} > ~/labops/evidence/fingerprints.txt


#View the file:
cat ~/labops/evidence/fingerprints.txt
