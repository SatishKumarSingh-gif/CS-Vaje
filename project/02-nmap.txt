#Run:
nmap -sV -p 2222 localhost


#Explanation:
#- nmap starts the Nmap scanning tool.
#- -sV attempts to identify the service version.
#- -p 2222 scans only port 2222.
#- localhost means your own computer.
#Possible output will look similar to:
PORT     STATE SERVICE VERSION
2222/tcp open  ssh     OpenSSH ...

nmap -sV -p 2222 localhost > ~/labops/evidence/02-nmap.txt


#View it:
cat ~/labops/evidence/02-nmap.txt
